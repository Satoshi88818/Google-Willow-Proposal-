    "## 3. Hamiltonian & Trotterization\n",
    "\n",
    "Target: 2D XXZ Heisenberg + transverse field (natural on square grid).\n",
    "\n",
    "H = Σ Jzz σz_i σz_j + Jxy (σx_i σx_j + σy_i σy_j) + Σ h_x σx_i"
   ]

   "source": [
    "# Hamiltonian parameters (tune for ergodicity vs integrability)\n",
    "J_zz = 1.0\n",
    "J_xy = 0.5\n",
    "h_x = 0.3\n",
    "delta_t = 0.25\n",
    "trotter_steps = 10   # total depth ~60–80 layers → fits Willow coherence\n",
    "\n",
    "def trotter_layer() -> cirq.Circuit:\n",
    "    c = cirq.Circuit()\n",
    "    # Horizontal bonds (even)\n",
    "    for i in rows:\n",
    "        for j in range(2, 10):\n",
    "            q1 = GridQubit(i, j)\n",
    "            q2 = GridQubit(i, j+1)\n",
    "            c.append(cirq.ISWAP(q1, q2) ** (J_xy * delta_t / np.pi))\n",
    "            c.append(cirq.CZ(q1, q2) ** (J_zz * delta_t / np.pi))\n",
    "    # Vertical bonds (odd)\n",
    "    for i in range(2, 10):\n",
    "        for j in cols:\n",
    "            q1 = GridQubit(i, j)\n",
    "            q2 = GridQubit(i+1, j)\n",
    "            c.append(cirq.ISWAP(q1, q2) ** (J_xy * delta_t / np.pi))\n",
    "            c.append(cirq.CZ(q1, q2) ** (J_zz * delta_t / np.pi))\n",
    "    # Transverse field\n",
    "    for q in qubits:\n",
    "        c.append(cirq.rx(h_x * delta_t * 2).on(q))\n",
    "    return c"
   ]

    "def build_echo_circuit(include_butterfly=True, nesting=1):\n",
    "    circuit = cirq.Circuit()\n",
    "    \n",
    "    # 1. Initial product state (all |0⟩ or optional |+⟩)\n",
    "    circuit.append(cirq.I.on_each(qubits))  # or cirq.H for |+⟩\n",
    "    \n",
    "    # 2. Forward evolution U(t)\n",
    "    for _ in range(trotter_steps):\n",
    "        circuit.append(trotter_layer())\n",
    "    \n",
    "    # 3. Butterfly perturbation (repeat for higher-order OTOC)\n",
    "    if include_butterfly:\n",
    "        for _ in range(nesting):\n",
    "            circuit.append(cirq.X(qb))\n",
    "            for _ in range(trotter_steps):\n",
    "                circuit.append(trotter_layer() ** -1)  # backward\n",
    "            for _ in range(trotter_steps):\n",
    "                circuit.append(trotter_layer())        # forward again\n",
    "    \n",
    "    # 4. Final backward evolution\n",
    "    for _ in range(trotter_steps):\n",
    "        circuit.append(trotter_layer() ** -1)\n",
    "    \n",
    "    # 5. Probe operator M = X on qm\n",
    "    circuit.append(cirq.X(qm))\n",
    "    \n",
    "    # 6. Measurement\n",
    "    circuit.append(cirq.measure(qm, key='echo'))\n",
    "    \n",
    "    return circuit\n",
    "\n",
    "# Generate base circuit (single nesting, with butterfly)\n",
    "echo_circuit = build_echo_circuit(include_butterfly=True, nesting=1)\n",

    "print(\"Number of qubits:\", len(qubits))\n",
    "print(\"Circuit ready for execution on Willow QVM or hardware.\")"
